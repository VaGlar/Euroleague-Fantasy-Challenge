// Baseline probe of functions/api/sync/[[path]].js against an in-memory D1 (nothing leaves the machine).
// usage: node sync_baseline.mjs <repo>
const repo = process.argv[2];
const mod = await import(repo + "/functions/api/sync/[[path]].js");
const { onRequest, newCode, cleanCode, ALPHABET, MAX_BYTES } = mod;

function fakeD1({ insertAlwaysIgnored = false, bumpBeforeUpdate = false } = {}) {
  const rows = new Map();
  const log = [];
  return { rows, log, prepare(sql) {
    const s = sql.replace(/\s+/g, " ").trim();
    return { args: [], bind(...a) { this.args = a; log.push({ sql: s, args: a.map((x) => typeof x) }); return this; },
      async run() { const a = this.args;
        if (s.startsWith("CREATE")) return { meta: { changes: 0 } };
        if (s.startsWith("INSERT OR IGNORE")) { if (insertAlwaysIgnored || rows.has(a[0])) return { meta: { changes: 0 } };
          rows.set(a[0], { data: a[1], rev: 1, updated_at: a[2] }); return { meta: { changes: 1 } }; }
        if (s.startsWith("UPDATE")) { const r = rows.get(a[2]);
          if (bumpBeforeUpdate) { r.rev += 1; r.data = JSON.stringify({ players: [], other: "device" }); }
          if (!r || r.rev !== a[3]) return { meta: { changes: 0 } };
          rows.set(a[2], { data: a[0], rev: r.rev + 1, updated_at: a[1] }); return { meta: { changes: 1 } }; }
        throw new Error("unexpected SQL " + s); },
      async first() { const r = rows.get(this.args[0]); return r ? { ...r } : null; } }; } };
}
async function call(db, method, path, body) {
  const parts = path.replace(/^\/api\/sync\/?/, "").split("/").filter(Boolean);
  const request = new Request("https://local.test" + path, { method,
    body: body === undefined ? undefined : typeof body === "string" ? body : JSON.stringify(body) });
  const r = await onRequest({ request, env: { SYNC_DB: db }, params: parts.length ? { path: parts } : {} });
  const text = await r.text();
  return { status: r.status, headers: Object.fromEntries(r.headers), body: text.slice(0, 160) };
}
const TEAM = { players: Array.from({ length: 11 }, (_, i) => ({ id: i + 1 })) };
const out = (name, v) => console.log(`## ${name}\n${JSON.stringify(v, null, 1)}\n`);

const created = await call(db, "POST", "/api/sync", { data: TEAM });
const code = JSON.parse(created.body).code;
out("POST new (status, response headers)", created);

out("DELETE on a code", await call(db, "DELETE", `/api/sync/${code}`));
out("OPTIONS on root", await call(db, "OPTIONS", "/api/sync"));

// code randomness: x % 31 over bytes 0..255 -> the first 256 % 31 = 8 letters are a bit more likely
const N = 31 * 20000 / 12 | 0, counts = Object.fromEntries([...ALPHABET].map((c) => [c, 0]));
for (let i = 0; i < N; i++) for (const ch of newCode()) counts[ch]++;
const total = N * 12, exp = total / 31;
out("newCode letter frequency vs uniform (ratio)", Object.fromEntries(Object.entries(counts).map(([c, n]) => [c, +(n / exp).toFixed(3)])));
out("entropy bits per code (with the modulo bias)", +(12 * -[...ALPHABET].reduce((a, c, i) => { const p = (i < 8 ? 9 : 8) / 256; return a + p * Math.log2(p); }, 0)).toFixed(2));
